# "scraper_jni": the robot driver with the C API and its JNI bridge, for the Kotlin apps
# (Android through the NDK, desktop through :shared:buildCoreJni).
# SCRAPER_DRIVER: the folder in drivers/ to build (needed when there is more than one).
cmake_minimum_required(VERSION 3.15)
project(scraper_jni LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(SCRAPER_DRIVER "" CACHE STRING "robot driver (a folder in drivers/)")
if(SCRAPER_DRIVER STREQUAL "")
  file(GLOB FOUND LIST_DIRECTORIES true ${CMAKE_CURRENT_LIST_DIR}/../drivers/*/src)
  list(LENGTH FOUND N)
  if(NOT N EQUAL 1)
    message(FATAL_ERROR "set SCRAPER_DRIVER to a folder in drivers/")
  endif()
  get_filename_component(FOUND ${FOUND} DIRECTORY)
  get_filename_component(SCRAPER_DRIVER ${FOUND} NAME)
endif()
get_filename_component(DRIVER_SRC ${CMAKE_CURRENT_LIST_DIR}/../drivers/${SCRAPER_DRIVER}/src ABSOLUTE)
file(GLOB DRIVER_SOURCES ${DRIVER_SRC}/*.cpp)
set(JNI_SOURCES ${CMAKE_CURRENT_LIST_DIR}/jni/scraper_jni.cpp)
if(NOT ANDROID)   # classic Bluetooth serial for the desktop app (Android has its own API)
  if(WIN32)
    set(BT_OS windows)
    set(BT_EXT .cpp)
  elseif(APPLE)
    set(BT_OS macos)
    set(BT_EXT .mm)   # IOBluetooth is Objective-C
    enable_language(OBJCXX)
  elseif(UNIX)
    set(BT_OS linux)
    set(BT_EXT .cpp)
  else()
    set(BT_OS none)
    set(BT_EXT .cpp)
  endif()
  list(APPEND JNI_SOURCES ${CMAKE_CURRENT_LIST_DIR}/jni/bt_jni.cpp ${CMAKE_CURRENT_LIST_DIR}/jni/bt_${BT_OS}${BT_EXT})
endif()
add_library(scraper_jni SHARED ${DRIVER_SOURCES} ${JNI_SOURCES})
target_include_directories(scraper_jni PRIVATE ${DRIVER_SRC} ${CMAKE_CURRENT_LIST_DIR})
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(scraper_jni PRIVATE ${JNI_INCLUDE_DIRS})
endif()
find_package(Threads REQUIRED)
target_link_libraries(scraper_jni PRIVATE Threads::Threads)
if(ANDROID)
  target_link_options(scraper_jni PRIVATE "-Wl,-z,max-page-size=16384")
elseif(UNIX AND NOT APPLE)
  # one self-contained .so for the desktop app
  target_link_options(scraper_jni PRIVATE -static-libstdc++ -static-libgcc)
elseif(MSVC)
  # the same on Windows: no separate C++ runtime to install
  set_property(TARGET scraper_jni PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
if(WIN32)
  target_link_libraries(scraper_jni PRIVATE ws2_32)
endif()
if(APPLE AND NOT ANDROID)
  set_source_files_properties(${CMAKE_CURRENT_LIST_DIR}/jni/bt_macos.mm PROPERTIES COMPILE_OPTIONS -fobjc-arc)
  target_link_libraries(scraper_jni PRIVATE "-framework IOBluetooth" "-framework Foundation")
endif()
